extends Node2D var server = WebSocketMultiplayerPeer.new() var port = 8080 # Variables del juego var pos_jugador = Vector2.ZERO var golpe_jugador = false var raqueta_color = "azul" func _ready(): # Iniciar servidor WebSocket compatible con Godot 4 var err = server.create_server(port) if err == OK: print("Servidor listo en la TV en el puerto: ", port) else: print("Error al iniciar servidor WebSocket: ", err) func _process(_delta): server.poll() if server.get_connection_status() == MultiplayerPeer.CONNECTION_CONNECTED: while server.get_available_packet_count() > 0: var packet = server.get_packet() var json_str = packet.get_string_from_utf8() _procesar_datos_celular(json_str) queue_redraw() # Redibuja la pantalla constantemente func _procesar_datos_celular(json_str): var json = JSON.new() if json.parse(json_str) == OK: var datos = json.get_data() var pantalla_ancho = get_viewport_rect().size.x var pantalla_alto = get_viewport_rect().size.y # Mapeamos la coordenada (0.0 a 1.0) enviada por el celular al tamaño de la pantalla pos_jugador.x = datos.get("x", 0.5) * pantalla_ancho pos_jugador.y = datos.get("y", 0.5) * pantalla_alto golpe_jugador = datos.get("golpe", 0) == 1 raqueta_color = datos.get("raqueta", "azul") # Dibujamos en pantalla el fondo negro y el jugador func _draw(): # Fondo Negro draw_rect(Rect2(Vector2.ZERO, get_viewport_rect().size), Color.BLACK) # Dibujar Jugador Stickman Blanco var cabeza_pos = pos_jugador - Vector2(0, 40) draw_circle(cabeza_pos, 20.0, Color.WHITE) # Cabeza draw_line(cabeza_pos, pos_jugador + Vector2(0, 50), Color.WHITE, 6.0) # Cuerpo # Dibujar Raqueta según color detectado var color_dibujo = Color.DODGER_BLUE if raqueta_color == "azul" else Color.ORANGE if golpe_jugador: draw_circle(pos_jugador + Vector2(30, -30), 25.0, Color.YELLOW) # Efecto de golpe draw_circle(pos_jugador + Vector2(30, -10), 15.0, color_dibujo)
